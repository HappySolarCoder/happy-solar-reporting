  function esc(value) {
    return String(value == null ? "" : value).replace(/[&<>"']/g, function (ch) {
      return ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" })[ch];
    });
  }

    return new Intl.DateTimeFormat("en-CA", {
      timeZone: "America/New_York", year: "numeric", month: "2-digit", day: "2-digit",
    }).format(new Date());
  }

    else url.searchParams.set("territory", next.territory);

  function sumKeys(obj, keys) {
    if (!obj || typeof obj !== "object") return null;
    const wants = keys.map(function (k) { return k.toLowerCase(); });
    let total = 0;
    Object.keys(obj).forEach(function (key) {
      if (wants.indexOf(String(key).toLowerCase()) >= 0) total += Number(obj[key] || 0);
    });
    return total;
  }
  function sumSweeper(obj) {
    if (!obj || typeof obj !== "object") return null;
    let total = 0;
    Object.keys(obj).forEach(function (key) {
      const low = String(key).toLowerCase();
      if (low === "sweeper" || low === "rehash") total += Number(obj[key] || 0);
    });
    return total;
  }
  function rateText(num, den) {
    if (num == null || den == null || Number.isNaN(Number(num)) || Number.isNaN(Number(den))) return "—";
    if (!Number(den)) return "—";
    return (100 * Number(num) / Number(den)).toFixed(1) + "%";
  }
  function numText(value) {
    if (value == null || Number.isNaN(Number(value))) return "—";
    return String(Math.round(Number(value)));
  }
  function daysInMonth(period) {
    return new Date(Date.UTC(Number(period.slice(0, 4)), Number(period.slice(5, 7)), 0)).getUTCDate();
  }
  function elapsedDays(period) {
    const today = nyToday();
    if (today.slice(0, 7) < period) return 0;
    if (today.slice(0, 7) > period) return daysInMonth(period);
    return Number(today.slice(8, 10));
  }

    const rows = bloom.territory_sales || [];
    for (let i = 0; i < rows.length; i += 1) {
      if (rows[i].territory === want && rows[i].target != null) return rows[i];

  }

  function demoPaceCaption(demos, ran, goal) {
    const parsed = Number(goal && goal.target);
    const target = Number.isFinite(parsed) ? parsed : COMPANY_DEMO_PCT_TARGET;
    if (demos == null || ran == null || !Number(ran) || Number.isNaN(Number(demos))) return "Goal " + target + "%";
    const shown = Number((100 * Number(demos) / Number(ran)).toFixed(1));
    const state = shown + 1e-9 >= target ? "On Track" : "Behind";
    return shown.toFixed(1) + "% vs " + target + "% target · " + state;
  }
  function paceCaption(actual, goal) {
    if (!goal || goal.target == null) return "Goal Not Set";
    const target = Number(goal.target);
    const period = goal.period_id;
    const days = daysInMonth(period);
    const elapsed = elapsedDays(period);
    if (actual == null || Number.isNaN(Number(actual))) return "Goal " + target + " · actual unavailable";
    const value = Number(actual);
    if (!elapsed) return value + " of " + target + " · month not started";
    const expected = target * elapsed / days;
    const state = value + 1e-9 >= expected ? "On Track" : "Behind";
    const locked = goal.locked_default ? " locked default" : "";
    return value + " of " + target + locked + " · day " + elapsed + " of " + days + " · " + state;
  }

  function recentMonths() {
    const today = nyToday();
    const endY = Number(today.slice(0, 4));
    const endM = Number(today.slice(5, 7));
    const months = [];
    let y = endY;
    let m = endM;
    for (let i = 0; i < 8; i += 1) {
      const id = y + "-" + String(m).padStart(2, "0");
      const start = id + "-01";
      const last = new Date(Date.UTC(y, m, 0)).getUTCDate();
      const end = id + "-" + String(last).padStart(2, "0");
      const partial = y === endY && m === endM;
      const label = new Intl.DateTimeFormat("en-US", { timeZone: "America/New_York", month: "short" }).format(new Date(Date.UTC(y, m - 1, 15, 17)));
      months.unshift({
        year: y, month: m, label: partial ? label + " MTD" : label,
        partial: partial, start: start, end: partial ? today : end, asOf: partial ? today : end,
      });
      m -= 1;
      if (m === 0) { m = 12; y -= 1; }
    }
    return months;
  }

  function channelsFrom(data) {
    const sales = data.sales || {};

    const demo = data.demo || {};

    const createdLead = (created.breakdowns || {}).created_by_lead_gen_source || null;

    const sitLead = (demo.breakdowns || {}).sit_by_lead_gen_source || null;

    const ranPipe = (demo.breakdowns || {}).ran_by_pipeline || (demo.breakdowns || {}).demo_rate_by_pipeline || null;
    const sitPipe = (demo.breakdowns || {}).sit_by_pipeline || null;
    const company = pack(
      created && created.error ? null : Number(created.result),
      demo && demo.error ? null : Number(demo.ran_count),
      demo && demo.error ? null : Number(demo.sit_count),
      sales && sales.error ? null : Number(sales.result)
    );
    function leadPack(keys) {
      return pack(sumKeys(createdLead, keys), sumKeys(ranLead, keys), sumKeys(sitLead, keys), sumKeys(salesLead, keys));
    }
    const sweeper = pack(sumSweeper(createdPipe), sumSweeper(ranPipe), sumSweeper(sitPipe), sumSweeper(salesPipe));
    return {
      Company: company,
      Doors: leadPack(DOOR_KEYS),
      "Self gen": data === snapshot ? (selfGenPack() || leadPack(SELF_KEYS)) : leadPack(SELF_KEYS),
      Sweeper: sweeper,
      Inbound: leadPack(INBOUND_KEYS),
      "3PL": leadPack(THREE_PL_KEYS),
      salesPipe: salesPipe,
      salesTotal: company.sales,
    };
  }

  function donut(pipe, total) {
    const host = document.getElementById("territoryDonut");
    if (!host) return;
    if (!pipe || total == null) {
      host.innerHTML = '<div class="pace-caption">Territory split unavailable</div>';
      return;
    }
    const entries = Object.keys(pipe).map(function (name) { return [name, Number(pipe[name] || 0)]; }).filter(function (row) { return row[1] > 0; });
    const sliceSum = entries.reduce(function (sum, row) { return sum + row[1]; }, 0);
    const colors = ["#26d9eb", "#2ea8fa", "#b59bff", "#24d4a6", "#ffbe45", "#ff8292"];
    let cursor = 0;
    const stops = entries.map(function (row, index) {
      const start = total ? (cursor / total) * 100 : 0;
      cursor += row[1];
      const end = total ? (cursor / total) * 100 : 0;
      return colors[index % colors.length] + " " + start + "% " + end + "%";
    });
    if (cursor < total) stops.push("#244559 " + (total ? (cursor / total) * 100 : 0) + "% 100%");
    const legend = entries.map(function (row, index) {
      return '<span><i style="background:' + colors[index % colors.length] + '"></i>' + esc(row[0]) + " <b>" + row[1] + "</b></span>";
    }).join("");
    const note = sliceSum === total ? "" : '<div class="pace-caption">Pipeline split ' + sliceSum + " vs sales total " + total + "</div>";
    host.innerHTML = '<div class="territory-content"><div class="ring" style="background:conic-gradient(' + (stops.join(",") || "#244559 0 100%") + ')" role="img" aria-label="' + esc(entries.map(function (row) { return row[0] + " " + row[1]; }).join(", ")) + '"><div class="ring-in">' + (total == null ? "—" : total) + '</div></div><div class="territory-legend">' + (legend || "<span>No sales in this scope</span>") + "</div></div>" + note;
  }

  }

  function seriesChart(rows, key, label, suffix) {
    const values = rows.filter(function (row) { return row[key] != null && !Number.isNaN(Number(row[key])); });
    if (!values.length) return '<p class="chart-note">Monthly trend unavailable.</p>';
    const max = Math.max(1, Math.max.apply(null, values.map(function (row) { return Number(row[key]); })));
    const x = 36, y = 12, w = 680, h = 160;
    const step = values.length > 1 ? w / (values.length - 1) : w;
    let path = "";
    const dots = values.map(function (row, i) {
      const px = x + i * step;
      const py = y + h - (Number(row[key]) / max) * h;
      path += (i ? "L" : "M") + px + "," + py + " ";
      return '<circle cx="' + px + '" cy="' + py + '" r="3.5" fill="#0b2439" stroke="#26d9eb" stroke-width="2"><title>' + esc(row.month) + " " + label + " " + row[key] + (suffix || "") + "</title></circle>";
    }).join("");
    const labels = values.map(function (row, i) {
      if (values.length > 8 && i % 2) return "";
      return '<text x="' + (x + i * step) + '" y="198" text-anchor="middle">' + esc(String(row.month).slice(2)) + "</text>";
    }).join("");
    return '<svg class="history-chart" viewBox="0 0 740 214" role="img" aria-label="' + esc(label) + ', monthly company totals"><path d="' + path + '" fill="none" stroke="#26d9eb" stroke-width="2.5" stroke-linejoin="round"/>' + dots + labels + "</svg>";
  }

  async function load() {
    const token = ++requestToken;
    pageFilters = filters();
    if (window.HappySolarLoading) window.HappySolarLoading.begin("view");
    const period = (pageFilters.start || "").slice(0, 7);
    bloomPending = true;
    bloomGoals = null;
    const bloomTask = fetch("/api/metrics/bloom_goals?oc_raw=1&period=" + encodeURIComponent(period), { hsOptional: true })
      .then(function (res) { return res && res.ok ? res.json() : null; })
      .catch(function (error) {
        if (error && error.name === "AbortError") throw error;
        return null;
      });
    channelGoals = null;
    const ym = (period || "").split("-");
    const channelTask = fetch("/api/metrics/channel_goals?year=" + encodeURIComponent(ym[0] || "") + "&month=" + encodeURIComponent(String(Number(ym[1] || 0))), { hsOptional: true })
      .then(function (res) { return res && res.ok ? res.json() : null; })
      .catch(function () { return null; });
    let response;
    try {
      response = await fetch("/api/metrics/company_snapshot?" + metricQuery().toString());
    } catch (error) {
      if (error && error.name === "AbortError") return;
      if (token !== requestToken) return;
      if (!snapshot) showUnavailable();
      return;
    }
    if (token !== requestToken) return;
    if (!response.ok) {
      if (!snapshot) showUnavailable();
      return;
    }
    let payload;
    try {
      payload = await response.json();
    } catch (error) {
      if (error && error.name === "AbortError") return;
      if (token !== requestToken) return;
      const generation = window.HappySolarLoading && window.HappySolarLoading.currentGeneration;
      if (generation && !generation.settled) generation.failed = true;
      if (!snapshot) showUnavailable();
      return;
    }
    if (token !== requestToken) return;
    applySnapshot(payload);
    paintChannelGoals(channelGoals ? null : "loading");
    channelTask.then(function (cg) {
      if (token !== requestToken) return;
      channelGoals = cg;
      if (cg && cg.self_gen && snapshot && selfGenPack()) fillStage("lgSelfGen", channelsFrom(snapshot)["Self gen"]);
      paintChannelGoals();
    });
    bloomTask.then(function (bloom) {
      if (token !== requestToken) return;
      bloomPending = false;
      bloomGoals = bloom || null;
      paintBloom();
    }).catch(function (error) {
      if (error && error.name === "AbortError") return;
      if (token !== requestToken) return;
      bloomPending = false;
      bloomGoals = null;
      paintBloom();
    });
  }

  function historyLines(rows, mode) {
    const config = mode === "counts"
      ? [["created", "Opps created", "#26d9eb"], ["ran", "Opps ran", "#2ea8fa"], ["demos", "Demos", "#b59bff"], ["sales", "Sales", "#24d4a6"]]
      : [["demoRate", "Demo %", "#26d9eb"], ["prelimRate", "Opp2Prelim", "#b59bff"]];
    const values = rows.filter(function (row) { return !row.missing; }).map(function (row) {
      return Object.assign({}, row, {
        demoRate: row.ran ? row.demos / row.ran * 100 : null,
        prelimRate: row.ran ? row.sales / row.ran * 100 : null,
      });
    });
    if (!values.length) return '<p class="chart-note">Monthly history unavailable.</p>';
    const max = mode === "rates" ? 100 : Math.max(4, Math.ceil(Math.max.apply(null, values.flatMap(function (row) {
      return config.map(function (item) { return row[item[0]] == null ? 0 : row[item[0]]; });
    })) / 4) * 4);
    const x = 45, y = 18, w = 680, h = 200, step = values.length > 1 ? w / (values.length - 1) : w;
    const demoTarget = Number(companyDemoGoal(bloomPending ? null : bloomGoals).target);
    const legend = '<div class="history-legend">' + config.map(function (item) {
      return '<span><i style="background:' + item[2] + '"></i>' + item[1] + "</span>";
    }).join("") + (mode === "rates" && Number.isFinite(demoTarget) ? '<span><i style="background:#ffbe45"></i>Demo % goal ' + demoTarget + "%</span>" : "") + "</div>";
    const grid = Array.from({ length: 5 }, function (_, i) {
      return '<line x1="' + x + '" x2="' + (x + w) + '" y1="' + (y + i * h / 4) + '" y2="' + (y + i * h / 4) + '" stroke="#29495f" stroke-dasharray="3 5"/><text x="' + (x - 10) + '" y="' + (y + i * h / 4 + 4) + '" text-anchor="end">' + (max - i * max / 4) + (mode === "rates" ? "%" : "") + "</text>";
    }).join("");
    const paths = config.map(function (item) {
      let path = "";
      let active = false;
      values.forEach(function (row, i) {
        const value = row[item[0]];
        if (value == null) { active = false; return; }
        path += (active ? "L" : "M") + (x + i * step) + "," + (y + h - value / max * h) + " ";
        active = true;
      });
      const dots = values.map(function (row, i) {
        if (row[item[0]] == null) return "";
        return '<circle cx="' + (x + i * step) + '" cy="' + (y + h - row[item[0]] / max * h) + '" r="4" fill="#0b2439" stroke="' + item[2] + '" stroke-width="2"><title>' + esc(row.month) + " · " + item[1] + ": " + Number(row[item[0]].toFixed(1)) + (mode === "rates" ? "%" : "") + "</title></circle>";
      }).join("");
      return '<path d="' + path + '" fill="none" stroke="' + item[2] + '" stroke-width="2.5" stroke-linejoin="round"/>' + dots;
    }).join("");
    const labels = values.map(function (row, i) {
      return '<text x="' + (x + i * step) + '" y="247" text-anchor="middle"' + (row.partial ? ' fill="#ffbe45"' : "") + ">" + esc(row.month) + "</text>";
    }).join("");
    let goalLine = "";
    if (mode === "rates" && Number.isFinite(demoTarget) && demoTarget > 0 && demoTarget <= max) {
      const gy = y + h - (demoTarget / max) * h;
      goalLine = '<line x1="' + x + '" x2="' + (x + w) + '" y1="' + gy + '" y2="' + gy + '" stroke="#ffbe45" stroke-width="1.5" stroke-dasharray="6 4"><title>Demo % goal ' + demoTarget + '%</title></line>';
    }
    const aria = mode === "counts" ? "Monthly totals" : ("Monthly conversion rates, Demo % goal " + demoTarget + "%");
    return legend + '<div class="history-chart-scroll"><svg class="history-chart" viewBox="0 0 760 265" role="img" aria-label="' + esc(aria) + '">' + grid + goalLine + paths + labels + "</svg></div>";
  }

  function bindCarousel() {
    const prev = document.getElementById("funnelPrev");
    const next = document.getElementById("funnelNext");
    const rail = document.getElementById("funnelCarousel");
    function visibleIndex() {
      if (!rail || window.innerWidth > 800) return 0;
      let closest = 0; let best = Infinity;
      Array.prototype.forEach.call(rail.children, function (card, i) {
        if (card.hidden) return;
        const delta = Math.abs(card.offsetLeft - rail.children[0].offsetLeft - rail.scrollLeft);
        if (delta < best) { best = delta; closest = i; }
      });
      return closest;
    }
    function dots() {
      document.querySelectorAll(".carousel-dot").forEach(function (dot, i) {
        const on = i === visibleIndex();
        dot.classList.toggle("active", on);
        dot.setAttribute("aria-pressed", String(on));
      });
    }
    if (prev) prev.addEventListener("click", function () { goFunnel(Math.max(0, visibleIndex() - 1)); });
    if (next) next.addEventListener("click", function () { goFunnel(Math.min(rail.children.length - 1, visibleIndex() + 1)); });
    document.querySelectorAll(".carousel-dot").forEach(function (dot) {
      dot.addEventListener("click", function () { goFunnel(Number(dot.getAttribute("data-index"))); });
    });
    if (rail) rail.addEventListener("scroll", dots);
  }
